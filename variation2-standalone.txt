  <script>
    const searchInput = document.getElementById('searchInput');
    const container = document.getElementById('container');
    const inputRow = document.getElementById('inputRow');
    const searchButton = document.getElementById('searchButton');
    const dropdown = document.getElementById('dropdown');
    const suggestionItems = document.querySelectorAll('.suggestion-item');

    let isOpen = false;

    // Focus handler - open dropdown
    searchInput.addEventListener('focus', function() {
      isOpen = true;
      container.style.background = '#ffffff';
      container.style.borderRadius = '24px';
      container.style.padding = '20px';
      container.style.boxShadow = '0 20px 25px -5px rgb(0 0 0 / 0.1), 0 8px 10px -6px rgb(0 0 0 / 0.1)';

      inputRow.style.gap = '0';

      searchInput.style.borderRadius = '16px';

      searchButton.style.width = '0';
      searchButton.style.height = '0';
      searchButton.style.opacity = '0';
      searchButton.style.overflow = 'hidden';

      dropdown.style.maxHeight = '600px';
      dropdown.style.opacity = '1';
      dropdown.style.marginTop = '16px';
    });

    // Blur handler - close dropdown
    searchInput.addEventListener('blur', function() {
      setTimeout(function() {
        isOpen = false;
        container.style.background = 'linear-gradient(to right, #93D7F9, #A4C1FF, #A3DEFF)';
        container.style.borderRadius = '50px';
        container.style.padding = '14px';
        container.style.boxShadow = 'none';

        inputRow.style.gap = '12px';

        searchInput.style.borderRadius = '50px';

        searchButton.style.width = '48px';
        searchButton.style.height = '48px';
        searchButton.style.opacity = '1';
        searchButton.style.overflow = 'visible';

        dropdown.style.maxHeight = '0';
        dropdown.style.opacity = '0';
        dropdown.style.marginTop = '0';
      }, 200);
    });

    // Hover effects for suggestions
    suggestionItems.forEach(function(item) {
      item.addEventListener('mouseenter', function() {
        this.style.backgroundColor = '#f9fafb';
      });

      item.addEventListener('mouseleave', function() {
        this.style.backgroundColor = 'transparent';
      });

      item.addEventListener('click', function() {
        searchInput.value = this.getAttribute('data-text');
      });
    });

    // Hover effect for search button
    searchButton.addEventListener('mouseenter', function() {
      this.style.backgroundColor = '#f9fafb';
    });

    searchButton.addEventListener('mouseleave', function() {
      this.style.backgroundColor = 'white';
    });
  </script>
